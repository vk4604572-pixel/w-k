    function safeJsonParse(key, fallback) {
      try {
        const raw = localStorage.getItem(key);
        return raw ? JSON.parse(raw) : fallback;
      } catch (e) {
        console.warn(`Invalid localStorage for "${key}"`, e);
        return fallback;
      }
    }

    function playSuccessSound() {
      try {
        const ctx = new (window.AudioContext || window.webkitAudioContext)();
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(587.33, ctx.currentTime);
        osc.frequency.setValueAtTime(880, ctx.currentTime + 0.12);
        gain.gain.setValueAtTime(0.2, ctx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.4);
        osc.connect(gain);
        gain.connect(ctx.destination);
        osc.start();
        osc.stop(ctx.currentTime + 0.45);
      } catch (e) {
        console.warn('Audio unavailable:', e);
      }
    }

    async function initWebcam() {
      const video = document.getElementById('webcam');
      const punchBtn = document.getElementById('btn-punch');

      if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
        showToast('This browser does not support camera access.', 'error');
        if (punchBtn) punchBtn.disabled = true;
        return;
      }

      try {
        const stream = await navigator.mediaDevices.getUserMedia({
          video: { facingMode: 'user', width: { ideal: 640 }, height: { ideal: 480 } },
          audio: false
        });

          video.srcObject = stream;

    function startSmartGPSWatch() {
      if (!('geolocation' in navigator)) {
        updateGPSUI(false, null, 'Geolocation not supported by device/browser');
        return;
      }

      const highAccuracyOptions = {
        enableHighAccuracy: true,
        timeout: 8000,
        maximumAge: 10000
      };

      navigator.geolocation.getCurrentPosition(
        onGPSSuccess,
        (error) => {
          console.warn('High accuracy GPS timeout, using fallback...', error);
          navigator.geolocation.getCurrentPosition(
            onGPSSuccess,
            handleGPSError,
            { enableHighAccuracy: false, timeout: 10000, maximumAge: 30000 }
          );
        },
        highAccuracyOptions
      );

      if (gpsWatchId) navigator.geolocation.clearWatch(gpsWatchId);
      gpsWatchId = navigator.geolocation.watchPosition(
        onGPSSuccess,
        () => {},
        { enableHighAccuracy: true, maximumAge: 15000, timeout: 12000 }
      );
    }

    function onGPSSuccess(pos) {
      userCurrentLocation = {
        lat: pos.coords.latitude,
        lng: pos.coords.longitude,
        accuracy: Math.round(pos.coords.accuracy)
      };

      const officeLocked = localStorage.getItem('smart_haziri_office_locked') === 'true';
      if (!officeLocked) {
        officeConfig = {
          lat: userCurrentLocation.lat,
          lng: userCurrentLocation.lng,
          radiusMeter: officeConfig.radiusMeter || 500
        };
        localStorage.setItem('smart_haziri_office', JSON.stringify(officeConfig));
        localStorage.setItem('smart_haziri_office_locked', 'true');
        updateOfficeDisplay();
      }

      updateGPSUI(true, userCurrentLocation);
    }

    function handleGPSError(err) {
      let msg = 'GPS Signal Error.';
      if (err.code === 1) {
        msg = "Permission Denied: Please tap 'Allow' for Location in browser lock icon.";
      } else if (err.code === 2) {
        msg = "Location unavailable. Turn ON 'High Accuracy' in phone settings.";
      } else if (err.code === 3) {
        msg = 'GPS Timeout: Weak satellite view. Move near a window or open sky.';
      }
      updateGPSUI(false, null, msg);
    }

      const statusText = document.getElementById('gps-status-text');

    }

    function calculateDistance(lat1, lon1, lat2, lon2) {
      const R = 6371e3;
      const toRad = deg => deg * (Math.PI / 180);
      const dLat = toRad(lat2 - lat1);
      const dLon = toRad(lon2 - lon1);
      const a =
        Math.sin(dLat / 2) * Math.sin(dLat / 2) +
        Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) *
        Math.sin(dLon / 2) * Math.sin(dLon / 2);

      const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
      return R * c;
    }

      renderTable();
      showToast(`Haajiri verified! ${matchedStaff.name} is marked present.`, 'success');
    }

    function captureRegistrationFace() {
      const video = document.getElementById('webcam');
      const canvas = document.getElementById('snapshotCanvas');

      if (!video || !canvas) {
        showToast('Camera unavailable. Please reload the page.', 'error');
        return;
      }

      if (!video.videoWidth || !video.videoHeight) {
        showToast('Camera is still loading. Please wait a moment.', 'info');
        return;
      }

      canvas.width = video.videoWidth || 640;
      canvas.height = video.videoHeight || 480;
      const ctx = canvas.getContext('2d');
      ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
      regFaceTemp = canvas.toDataURL('image/jpeg', 0.85);

      const regPreview = document.getElementById('reg-preview');
      const regPlaceholder = document.getElementById('reg-preview-placeholder');
      if (regPreview) {
        regPreview.src = regFaceTemp;
        regPreview.classList.remove('hidden');
      }
      if (regPlaceholder) regPlaceholder.classList.add('hidden');

      showToast('Face frame captured successfully!', 'success');
    }

    function handleStaffEnrollment(e) {
      e.preventDefault();

      if (!regFaceTemp) {
        showToast('Kripya pehle face capture karein!', 'error');
        return;
      }

      const name = document.getElementById('reg-name').value.trim();
      const empid = document.getElementById('reg-empid').value.trim();
      const role = document.getElementById('reg-role').value.trim() || 'Staff';

      if (!name || !empid) {
        showToast('Name and Employee ID are required.', 'error');
        return;
      }

      const staffObj = {
        name,
        empid,
        role,
        photo: regFaceTemp
      };

      registeredStaff = [staffObj, ...registeredStaff];
      localStorage.setItem('smart_haziri_staff', JSON.stringify(registeredStaff));

      showToast(`${staffObj.name} successfully registered!`, 'success');

      e.target.reset();
      regFaceTemp = null;

      const regPreview = document.getElementById('reg-preview');
      const regPlaceholder = document.getElementById('reg-preview-placeholder');
      if (regPreview) regPreview.classList.add('hidden');
      if (regPlaceholder) regPlaceholder.classList.remove('hidden');

      switchTab('scanner');
    }

    function renderTable() {
      const tbody = document.getElementById('attendance-table-body');
      const noMsg = document.getElementById('no-records-msg');

      if (!tbody) return;

      if (attendanceRecords.length === 0) {
        tbody.innerHTML = '';
        if (noMsg) noMsg.classList.remove('hidden');
        return;
      }

      if (noMsg) noMsg.classList.add('hidden');

      tbody.innerHTML = attendanceRecords.map(item => `
        <tr class="hover:bg-slate-800/40 transition">
          <td class="py-2.5 px-4">
            <img src="${item.photo}" class="w-9 h-9 rounded-xl object-cover border border-slate-700">
          </td>
          <td class="py-2.5 px-4">
            <div class="font-bold text-white">${item.name}</div>
            <div class="text-[10px] text-slate-500">${item.role} (${item.staffId})</div>
          </td>
          <td class="py-2.5 px-4">
            <div class="text-slate-200">${item.timestamp}</div>
            <div class="text-[10px] text-slate-500">${item.date}</div>
          </td>
          <td class="py-2.5 px-4">
            <a href="https://maps.google.com/?q=${item.lat},${item.lng}" target="_blank" class="text-emerald-400 hover:underline flex items-center gap-1 font-mono text-[11px]">
              ${item.lat.toFixed(4)}, ${item.lng.toFixed(4)}
              <i data-lucide="external-link" class="w-3 h-3"></i>
            </a>
            <div class="text-[10px] text-slate-500">${item.distance}m from site</div>
          </td>
          <td class="py-2.5 px-4">
            <span class="px-2 py-0.5 rounded-full text-[10px] font-bold ${
              item.status === 'On-Site'
                ? 'bg-emerald-500/10 text-emerald-400 border border-emerald-500/30'
                : 'bg-amber-500/10 text-amber-400 border border-amber-500/30'
            }">
              ${item.status}
            </span>
          </td>
        </tr>
      `).join('');

      lucide.createIcons();
    }

    function exportToCSV() {
      if (attendanceRecords.length === 0) {
        showToast('No records to export!', 'info');
        return;
      }

      let csv = 'Employee,EmpID,Role,Date,Time,Latitude,Longitude,DistanceMeter,Status\n';
      attendanceRecords.forEach(r => {
        csv += `"${r.name}","${r.staffId}","${r.role}","${r.date}","${r.timestamp}",${r.lat},${r.lng},${r.distance},"${r.status}"\n`;
      });

      const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `attendance_${new Date().toISOString().slice(0, 10)}.csv`;
      a.click();
      URL.revokeObjectURL(url);
    }

    function switchTab(tab) {
      const tabs = ['scanner', 'enroll', 'logs'];
      tabs.forEach(name => {
        const view = document.getElementById(`view-${name}`);
        if (view) view.classList.add('hidden');
        const btn = document.getElementById(`tab-${name}`);
        if (btn) {
          btn.className = 'px-3 sm:px-4 py-1.5 rounded-lg font-medium transition-all text-slate-400 hover:text-white';
        }
      });

      if (tab === 'scanner') {
        const view = document.getElementById('view-scanner');
        const btn = document.getElementById('tab-scanner');
        if (view) view.classList.remove('hidden');
        if (btn) btn.className = 'px-3 sm:px-4 py-1.5 rounded-lg font-medium transition-all bg-emerald-500 text-slate-950 shadow';
      }

      if (tab === 'enroll') {
        const view = document.getElementById('view-enroll');
        const btn = document.getElementById('tab-enroll');
        if (view) view.classList.remove('hidden');
        if (btn) btn.className = 'px-3 sm:px-4 py-1.5 rounded-lg font-medium transition-all bg-emerald-500 text-slate-950 shadow';
      }

      if (tab === 'logs') {
        const view = document.getElementById('view-logs');
        const btn = document.getElementById('tab-logs');
        if (view) view.classList.remove('hidden');
        if (btn) btn.className = 'px-3 sm:px-4 py-1.5 rounded-lg font-medium transition-all bg-emerald-500 text-slate-950 shadow';
      }
    }

    function showToast(message, type = 'info') {
      const toast = document.getElementById('toast');
      if (!toast) return;

      toast.innerText = message;

      if (type === 'success') {
        toast.className = 'fixed bottom-6 right-6 z-50 transform translate-y-0 opacity-100 transition-all duration-300 max-w-sm px-4 py-3 rounded-2xl shadow-2xl border text-xs bg-emerald-950 border-emerald-500 text-emerald-200';
      } else if (type === 'error') {
        toast.className = 'fixed bottom-6 right-6 z-50 transform translate-y-0 opacity-100 transition-all duration-300 max-w-sm px-4 py-3 rounded-2xl shadow-2xl border text-xs bg-rose-950 border-rose-500 text-rose-200';
      } else {
        toast.className = 'fixed bottom-6 right-6 z-50 transform translate-y-0 opacity-100 transition-all duration-300 max-w-sm px-4 py-3 rounded-2xl shadow-2xl border text-xs bg-slate-900 border-slate-700 text-slate-200';
      }

      clearTimeout(showToast.timeoutId);
      showToast.timeoutId = setTimeout(() => {
        toast.className = 'fixed bottom-6 right-6 z-50 transform translate-y-20 opacity-0 transition-all duration-300 max-w-sm px-4 py-3 rounded-2xl shadow-2xl border text-xs';
      }, 3500);
    }
