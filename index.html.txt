    doc.setFont("helvetica","normal");doc.setFontSize(10);doc.text(R.fl+" · As on "+R.t0,36,52);

function rdB(f){
  if(!f)return;const r=new FileReader();
  r.onload=()=>{
    try{
      const b=JSON.parse(r.result);
      if(Array.isArray(b.V))V=b.V;if(Array.isArray(b.D))D=b.D;if(Array.isArray(b.E))E=b.E;if(Array.isArray(b.Sv))Sv=b.Sv;
      if(Array.isArray(b.FIRMS))FIRMS=b.FIRMS;if(Array.isArray(b.CUST))CUST=b.CUST;
      saveLocal();render();alert("Backup restored successfully!");
    }catch(e){alert("Invalid backup file");}
  };
  r.readAsText(f);
}
